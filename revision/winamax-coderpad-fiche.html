Exo D — updateProgress(list, id, progress)

≤0 unread · >0 et <100 watching · ≥100 seen (clamp 0..100)

Ta solution
function updateProgress(list, id, progress) {
  return list.map(c => {
    if (c.id !== id) return c;
    const p = Math.max(0, Math.min(100, progress));
    const status = p <= 0 ? "unread" : p >= 100 ? "seen" : "watching";
    return { ...c, progress: p, status };
  });
}

Exo I — Mini lecteur

State playing + progress. Boutons Play / Pause / +10% / Reset. À 100% → seen.

Ta solution
function Player() {
  const [playing, setPlaying] = useState(false);
  const [progress, setProgress] = useState(0);
  return (
    <div>
      <p>{playing ? "Lecture" : "Pause"} — {progress}%</p>
      <button onClick={() => setPlaying(true)}>Play</button>
      <button onClick={() => setPlaying(false)}>Pause</button>
      <button onClick={() => setProgress(p => Math.min(100, p + 10))}>+10%</button>
    </div>
  );
}

function createContent(title) {
  const item = { id: Date.now(), title, status: "unread", progress: 0 };
  contents.push(item);
  return item;
